**Update `src/app/payments/[id]/receipt/page.tsx`:**

```tsx
'use client';

import { useEffect, useState } from 'react';
import { useParams } from 'next/navigation';
import { ReceiptView, PaymentWithRelations } from '@/components/payment/receipt-view';

export default function ReceiptPage() {
  const params = useParams();
  const [payment, setPayment] = useState<PaymentWithRelations | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    fetchPayment();
  }, [params.id]);

  async function fetchPayment() {
    try {
      const res = await fetch(`/api/v1/payments/${params.id}`);
      if (!res.ok) throw new Error('Failed to fetch payment');
      const { data } = await res.json();
      setPayment(data);
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Unknown error');
    } finally {
      setLoading(false);
    }
  }

  if (loading) {
    return (
      <div className="flex items-center justify-center min-h-[200px]">
        <p className="text-muted-foreground">Loading receipt...</p>
      </div>
    );
  }

  if (error || !payment) {
    return (
      <div className="flex items-center justify-center min-h-[200px]">
        <p className="text-red-500">{error || 'Payment not found'}</p>
      </div>
    );
  }

  return (
    <div className="container py-6">
      {/* Page header - hidden in print */}
      <div className="mb-6 no-print">
        <h1 className="text-2xl font-semibold">Payment Receipt</h1>
        <p className="text-muted-foreground mt-1">
          Receipt #{payment.referenceNumber || 'N/A'}
        </p>
      </div>

      {/* Receipt with buttons */}
      <ReceiptView
        payment={payment}
        showButtons={true}
        className="max-w-2xl mx-auto"
      />
    </div>
  );
}
```
