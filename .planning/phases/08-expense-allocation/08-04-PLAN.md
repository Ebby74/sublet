## Task 4.1: Add Expense Allocation to Business Summary Report

**File:** src/components/reports/business-summary-report.tsx

Add new section:

```tsx
{/* Expense Allocation Section */}
<section className="border rounded-lg p-4">
  <h3 className="text-lg font-semibold mb-4">Expense Allocation</h3>
  
  {/* Summary Cards */}
  <div className="grid grid-cols-4 gap-4 mb-4">
    {expenseAllocation.map((source) => (
      <div key={source.value} className="p-3 rounded-lg border">
        <p className={`text-sm font-medium ${source.color}`}>
          {source.label}
        </p>
        <p className="text-xl font-bold">
          RM {(source.total / 100).toLocaleString()}
        </p>
        <p className="text-sm text-muted">
          {source.percentage}% of total
        </p>
      </div>
    ))}
  </div>
  
  {/* Breakdown Chart */}
  <ResponsiveContainer width="100%" height={200}>
    <PieChart>
      <Pie
        data={expenseAllocation}
        dataKey="total"
        nameKey="label"
        cx="50%"
        cy="50%"
        outerRadius={80}
        label={({label, percentage}) => `${label}: ${percentage}%`}
      >
        {expenseAllocation.map((entry, index) => (
          <Cell key={entry.value} fill={COLORS[index % COLORS.length]} />
        ))}
      </Pie>
    </PieChart>
  </ResponsiveContainer>
  
  {/* Unallocated Alert */}
  {unallocatedCount > 0 && (
    <div className="mt-4 p-3 bg-yellow-50 border border-yellow-200 rounded">
      <p className="text-sm text-yellow-800">
        <strong>Note:</strong> {unallocatedCount} expenses ({formatCurrency(unallocatedTotal)}) 
        are not yet tagged to an income source. 
        <button className="underline ml-1">Tag now</button>
      </p>
    </div>
  )}
</section>
```
