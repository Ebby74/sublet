case 'expenses': {
  const payments = await getPayments(userId, { type: 'expense' });
  const data = payments.map(p => ({
    'Date': formatDate(p.paidAt || p.dueDate),
    'Description': p.description,
    'Category': p.category,
    'Amount (RM)': p.amountSen / 100,  // Convert from sen

### Add 'expense-allocation' export type

New export for expense allocation report:

```typescript
case 'expense-allocation': {
  const payments = await getPayments(userId, { type: 'expense' });
  
  // Group by income source
  const bySource = groupBy(payments, 'incomeSource');
  
  const sheets = [
    {
      name: 'By Income Source',
      data: Object.entries(bySource).map(([source, items]) => ({
        'Income Source': INCOME_SOURCES.find(s => s.value === source)?.label || 'Unallocated',
        'Total (RM)': items.reduce((sum, p) => sum + p.amountSen, 0) / 100,
        'Count': items.length,
        'Percentage': (items.reduce((sum, p) => sum + p.amountSen, 0) / 
                      payments.reduce((sum, p) => sum + p.amountSen, 0) * 100).toFixed(1) + '%',
      }))
    },
    {
      name: 'All Expenses',
      data: payments.map(p => ({
        'Date': formatDate(p.paidAt || p.dueDate),
        'Description': p.description,
        'Category': p.category,
        'Amount (RM)': p.amountSen / 100,
        'Income Source': INCOME_SOURCES.find(s => s.value === p.incomeSource)?.label || 'Unallocated',
      }))
    }
  ];
  
  return buildExcelWorkbook(sheets);
}
```
