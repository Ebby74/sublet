## Task 2.1: Create Income Source Constants

**File:** src/lib/income-sources.ts (new)

```typescript
export type IncomeSource = 'sublet' | 'autoren_sell' | 'autoren_rent' | 'unallocated';

export const INCOME_SOURCES: { value: IncomeSource; label: string; color: string }[] = [
  { value: 'sublet', label: 'Sublet', color: 'bg-blue-100 text-blue-800' },
  { value: 'autoren_sell', label: 'Autoren Sell', color: 'bg-green-100 text-green-800' },
  { value: 'autoren_rent', label: 'Autoren Rent', color: 'bg-purple-100 text-purple-800' },
  { value: 'unallocated', label: 'Unallocated', color: 'bg-gray-100 text-gray-800' },
];

// Smart categorization rules
export const CATEGORY_TO_INCOME_SOURCE: Record<string, IncomeSource> = {
  // Sublet categories
  'property_maintenance': 'sublet',
  'utilities': 'sublet',
  'property_renovation': 'sublet',
  'security': 'sublet',
  'cleaning': 'sublet',
  'repairs': 'sublet',
  
  // Autoren Rent categories
  'marketing': 'autoren_rent',
  'advertising': 'autoren_rent',
  'photography': 'autoren_rent',
  'listing_fees': 'autoren_rent',
  
  // Autoren Sell categories
  'legal_fees': 'autoren_sell',
  'stamp_duty': 'autoren_sell',
  'valuation_fees': 'autoren_sell',
  
  // Shared categories (default to unallocated)
  'office_supplies': 'unallocated',
  'software': 'unallocated',
  'travel': 'unallocated',
  'transport': 'unallocated',
  'bank_fees': 'unallocated',
  'communication': 'unallocated',
  'other': 'unallocated',
};

export function getSuggestedIncomeSource(category: string): IncomeSource {
  return CATEGORY_TO_INCOME_SOURCE[category.toLowerCase()] || 'unallocated';
}
```

## Task 2.2: Update Expense Form

**File:** src/components/payments/expense-form.tsx (or payment-form.tsx)

Add income source selection:

```tsx
interface ExpenseFormProps {
  // ... existing props
  onIncomeSourceChange?: (source: IncomeSource) => void;
}

// Add to form:
<div className="mb-4">
  <label className="block text-sm font-medium mb-1">Income Source</label>
  <select
    value={formData.incomeSource || 'unallocated'}
    onChange={(e) => {
      setFormData({ ...formData, incomeSource: e.target.value as IncomeSource });
      onIncomeSourceChange?.(e.target.value as IncomeSource);
    }}
    className="w-full p-2 border rounded-md"
  >
    {INCOME_SOURCES.map((source) => (
      <option key={source.value} value={source.value}>
        {source.label}
      </option>
    ))}
  </select>
  {suggestedSource && suggestedSource !== 'unallocated' && (
    <p className="text-sm text-muted mt-1">
      Suggested: {INCOME_SOURCES.find(s => s.value === suggestedSource)?.label}
    </p>
  )}
</div>
```

## Task 2.3: Add Smart Suggestion Logic

On category change, suggest income source:

```tsx
const [suggestedSource, setSuggestedSource] = useState<IncomeSource | null>(null);

useEffect(() => {
  if (formData.category) {
    const suggested = getSuggestedIncomeSource(formData.category);
    setSuggestedSource(suggested);
  }
}, [formData.category]);
```

Show suggestion badge when different from current selection.
