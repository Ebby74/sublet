    ```typescript
    function normalizePhoneNumber(phone: string): string {
      const digits = phone.replace(/[^\d+]/g, '');
      if (phone.startsWith('+')) {
        return digits;
      }
      if (digits.startsWith('0')) {
        return `+60${digits.slice(1)}`;
      }
      return `+60${digits}`;
    }
    ```

    2. **createViewing** - Creates viewing, validates room is active, sends WhatsApp confirmation:
    ```typescript
    import { prisma } from '@/lib/prisma';
    import { whatsappService } from './whatsapp-service';
    import { format } from 'date-fns';

    export interface CreateViewingInput {
      roomId: string;
      prospectId: string;
      scheduledAt: Date;
    }

    export async function createViewing(input: CreateViewingInput, userId: string) {
      // Verify room exists and is active
      const room = await prisma.room.findUnique({
        where: { id: input.roomId },
        include: { property: true },
      });

      if (!room || room.status !== 'active') {
        throw new Error('Room not available for viewing');
      }

      // Get prospect phone
      const prospect = await prisma.prospect.findUnique({
        where: { id: input.prospectId },
      });

      if (!prospect) {
        throw new Error('Prospect not found');
      }

      // Create viewing
      const viewing = await prisma.viewing.create({
        data: {
          roomId: input.roomId,
          prospectId: input.prospectId,
          scheduledAt: input.scheduledAt,
          status: 'scheduled',
        },
        include: {
          room: { include: { property: true } },
          prospect: true,
        },
      });

      // Send WhatsApp confirmation if phone exists
      if (prospect.phone) {
        const message = `🏠 Viewing Confirmed!\n\n` +
          `Room: ${viewing.room.name}\n` +
          `Property: ${viewing.room.property.address}\n` +
          `Date: ${format(viewing.scheduledAt, 'dd/MM/yyyy')}\n` +
          `Time: ${format(viewing.scheduledAt, 'hh:mm a')}\n\n` +
          `Reply CANCEL to reschedule`;

        const config = await marketingChannelService.getChannelConfig(userId, 'whatsapp');
        if (config?.whatsapp?.senderNumber) {
          await whatsappService.sendMessage(
            normalizePhoneNumber(prospect.phone),
            message,
            config.whatsapp.senderNumber
          );
        }
      }

      return viewing;
    }
    ```

    3. **getViewings** - Admin list viewings with filters:
    ```typescript
    export async function getViewings(userId: string, filters?: { roomId?: string; status?: string; from?: Date; to?: Date }) {
      const where: Record<string, unknown> = {
        room: { property: { userId } },
      };
      if (filters?.roomId) where.roomId = filters.roomId;
      if (filters?.status) where.status = filters.status;
      if (filters?.from || filters?.to) {
        where.scheduledAt = {
          gte: filters.from,
          lte: filters.to,
        };
      }

      return prisma.viewing.findMany({
        where,
        include: {
          room: { include: { property: true } },
          prospect: true,
        },
        orderBy: { scheduledAt: 'asc' },
      });
    }
    ```

    ```typescript
    import { createViewing, getViewings } from '@/services/viewing-service';

    const createViewingSchema = z.object({
      roomId: z.string().uuid(),
      prospectId: z.string().uuid(),
      scheduledAt: z.string().datetime(),
    });

    export async function GET(request: NextRequest) {
      try {
        const userId = 'demo-user'; // TODO: get from session
        const { searchParams } = new URL(request.url);
        const filters = {
          roomId: searchParams.get('roomId') || undefined,
          status: searchParams.get('status') || undefined,
        };

        const viewings = await getViewings(userId, filters);
        return NextResponse.json({ data: viewings, meta: { total: viewings.length } });
      } catch (error) {
        return NextResponse.json({ error: 'Failed to fetch viewings' }, { status: 500 });
      }
    }

    export async function POST(request: NextRequest) {
      try {
        const userId = 'demo-user';
        const body = await request.json();
        const validated = createViewingSchema.parse(body);

            roomId: validated.roomId,
            prospectId: validated.prospectId,
            scheduledAt: new Date(validated.scheduledAt),

        return NextResponse.json({ data: viewing }, { status: 201 });
    ```

    **2. Single viewing route** (`src/app/api/v1/viewings/[id]/route.ts`):
    ```typescript
    import { NextRequest, NextResponse } from 'next/server';
    import { completeViewing, cancelViewing } from '@/services/viewing-service';
    import { z } from 'zod';

    const updateViewingSchema = z.object({
      status: z.enum(['scheduled', 'completed', 'cancelled']).optional(),
      result: z.enum(['interested', 'not_interested']).optional(),
      notes: z.string().optional(),
    });

    export async function GET(
      request: NextRequest,
      { params }: { params: { id: string } }
    ) {
      // Get single viewing
      const { getViewing } = await import('@/services/viewing-service');
      const viewing = await getViewing(params.id);
      if (!viewing) {
        return NextResponse.json({ error: 'Viewing not found' }, { status: 404 });
      }
      return NextResponse.json({ data: viewing });
    }

    export async function PATCH(
      request: NextRequest,
      { params }: { params: { id: string } }
    ) {
      try {
        const body = await request.json();
        const validated = updateViewingSchema.parse(body);

        if (validated.status === 'completed' && validated.result) {
          const viewing = await completeViewing(params.id, validated.result, validated.notes);
          return NextResponse.json({ data: viewing });
        }

        if (validated.status === 'cancelled') {
          const viewing = await cancelViewing(params.id);
          return NextResponse.json({ data: viewing });
        }

        return NextResponse.json({ error: 'Invalid update' }, { status: 400 });
      } catch (error) {
        return NextResponse.json({ error: 'Failed to update viewing' }, { status: 500 });
      }
    }
    ```

    ```typescript
    import { NextRequest, NextResponse } from 'next/server';
    import { createViewing } from '@/services/viewing-service';
    import { z } from 'zod';

    const bookViewingSchema = z.object({
      roomId: z.string().uuid(),
      prospectName: z.string().min(1),
      prospectPhone: z.string().min(10),
      scheduledAt: z.string().datetime(),
    });

    export async function POST(request: NextRequest) {
      try {
        const body = await request.json();
        const validated = bookViewingSchema.parse(body);

        // Get or create prospect by phone
        const { prisma } = await import('@/lib/prisma');
        let prospect = await prisma.prospect.findFirst({
          where: { phone: validated.prospectPhone },
        });

        if (!prospect) {
          prospect = await prisma.prospect.create({
            data: {
              name: validated.prospectName,
              phone: validated.prospectPhone,
              source: 'website',
              status: 'viewing_scheduled',
              roomId: validated.roomId,
            },
          });
        }

        // Create viewing (userId from environment for public route)
        const viewing = await createViewing(
          {
            roomId: validated.roomId,
            prospectId: prospect.id,
            scheduledAt: new Date(validated.scheduledAt),
          },
          process.env.SYSTEM_USER_ID || 'system'
        );

        return NextResponse.json({
          data: viewing,
          meta: { message: 'Viewing booked successfully' }
        }, { status: 201 });
      } catch (error) {
        if (error instanceof z.ZodError) {
          return NextResponse.json({ error: error.errors }, { status: 400 });
        }
        return NextResponse.json({ error: 'Failed to book viewing' }, { status: 500 });
      }
    }
    ```
